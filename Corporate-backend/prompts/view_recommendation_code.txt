def plot_financial_indicators(data: pd.DataFrame):
    data['日期'] = pd.to_datetime(data['年份'], format='%Y')

    roe_chart = alt.Chart(data).mark_line(color='brown').encode(
        x='日期:T',
        y='净资产收益率:Q',
        tooltip=['日期:T', '净资产收益率:Q']
    ).properties(
        title='净资产收益率',
        width='container',
        height='container'
    )

    net_margin_chart = alt.Chart(data).mark_line(color='magenta').encode(
        x='日期:T',
        y='销售净利率:Q',
        tooltip=['日期:T', '销售净利率:Q']
    ).properties(
        title='销售净利率',
        width='container',
        height='container'
    )

    asset_turnover_chart = alt.Chart(data).mark_line(color='cyan').encode(
        x='日期:T',
        y='资产周转率:Q',
        tooltip=['日期:T', '资产周转率:Q']
    ).properties(
        title='资产周转率',
        width='container',
        height='container'
    )

    chart = alt.layer(roe_chart, net_margin_chart, asset_turnover_chart).resolve_scale(
        y='independent'
    ).properties(
        width='container',
        height='container'
    )
    
    return chart

def plot_stacked_bar_chart(data: pd.DataFrame):
    # 准备数据，假设要展示多个列的堆叠情况，比如流动资产合计和非流动资产合计
    data_melted = data.melt(id_vars=['年份'], value_vars=['流动资产合计', '非流动资产合计'], 
                            var_name='资产类型', value_name='资产值')

    # 创建堆叠条形图
    stacked_bar_chart = alt.Chart(data_melted).mark_bar().encode(
        y=alt.Y('年份:N', axis=alt.Axis(title='年份')),
        x=alt.X('资产值:Q', axis=alt.Axis(title='资产值')),
        color=alt.Color('资产类型:N', legend=alt.Legend(title='资产类型')),
        tooltip=['年份:N', '资产类型:N', '资产值:Q']
    ).properties(
        title='流动资产与非流动资产的堆叠条形图',
        width='container',
        height='container'
    )

    return stacked_bar_chart
            """
        }

def plot_pie_chart(data: pd.DataFrame):
    # 准备数据
    pie_data = data[['年份', '资产总计']]
    pie_data['资产总计百分比'] = pie_data['资产总计'] / pie_data['资产总计'].sum() * 100  # 计算百分比

    # 创建饼图
    pie_chart = alt.Chart(pie_data).mark_arc().encode(
        theta=alt.Theta(field="资产总计百分比", type="quantitative"),
        color=alt.Color(field="年份", type="nominal", legend=alt.Legend(title="年份")),
        tooltip=[alt.Tooltip('年份', title='年份'), alt.Tooltip('资产总计百分比', title='资产总计百分比', format='.2f')]
    ).properties(
        title='各年份资产总计分布情况',
        width='container',
        height='container'
    )

    return pie_chart
            """
        }

def plot_donut_chart(data: pd.DataFrame):
    # 准备数据
    pie_data = data[['年份', '资产总计']]
    pie_data['资产总计百分比'] = pie_data['资产总计'] / pie_data['资产总计'].sum() * 100  # 计算百分比

    # 创建环形图
    donut_chart = alt.Chart(pie_data).mark_arc(innerRadius=50).encode(
        theta=alt.Theta(field="资产总计百分比", type="quantitative"),
        color=alt.Color(field="年份", type="nominal", legend=alt.Legend(title="年份")),
        tooltip=[alt.Tooltip('年份', title='年份'), alt.Tooltip('资产总计百分比', title='资产总计百分比', format='.2f')]
    ).properties(
        title='各年份资产总计分布环形图',
        width='container',
        height='container'
    )

    # 添加文本标签
    text = donut_chart.mark_text(radiusOffset=10).encode(
        text=alt.Text(field="资产总计百分比", type="quantitative", format='.2f')
    )

    # 合并图表和文本
    final_chart = donut_chart + text
    final_chart = final_chart.configure_title(
        fontSize=20,
        anchor='middle'
    ).configure_legend(
        titleFontSize=14,
        labelFontSize=12
    )

    return final_chart
            """
        }
